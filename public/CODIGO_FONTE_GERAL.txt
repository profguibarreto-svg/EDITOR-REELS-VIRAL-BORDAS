// Servidor local puro em Node.js (compatível com Node 14, 16, 18, 20, 22+)
// Sem dependência externa de bundler ou npm packages!
import http from 'http';
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const DIST_DIR = path.join(__dirname, 'dist');
const PORT = process.env.PORT || 3000;

const MIME_TYPES = {
  '.html': 'text/html; charset=UTF-8',
  '.js': 'application/javascript; charset=UTF-8',
  '.mjs': 'application/javascript; charset=UTF-8',
  '.css': 'text/css; charset=UTF-8',
  '.json': 'application/json',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.jpeg': 'image/jpeg',
  '.svg': 'image/svg+xml',
  '.ico': 'image/x-icon',
  '.mp4': 'video/mp4',
  '.webm': 'video/webm',
  '.zip': 'application/zip',
  '.woff2': 'font/woff2',
  '.woff': 'font/woff',
  '.ttf': 'font/ttf',
};

const server = http.createServer((req, res) => {
  // CORS & Security headers for high-performance canvas
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Cross-Origin-Opener-Policy', 'same-origin');

  let reqPath = decodeURI(req.url.split('?')[0]);
  if (reqPath === '/') reqPath = '/index.html';

  let filePath = path.join(DIST_DIR, reqPath);

  // Normalize path security
  if (!filePath.startsWith(DIST_DIR)) {
    res.writeHead(403);
    res.end('Acesso proibido');
    return;
  }

  // Fallback to index.html for SPA routing if file doesn't exist
  if (!fs.existsSync(filePath) || fs.statSync(filePath).isDirectory()) {
    filePath = path.join(DIST_DIR, 'index.html');
  }

  const ext = path.extname(filePath).toLowerCase();
  const contentType = MIME_TYPES[ext] || 'application/octet-stream';

  fs.readFile(filePath, (err, data) => {
    if (err) {
      res.writeHead(500);
      res.end(`Erro ao carregar arquivo: ${err.message}`);
      return;
    }
    res.writeHead(200, { 'Content-Type': contentType });
    res.end(data);
  });
});

server.listen(PORT, '0.0.0.0', () => {
  console.log(`\n✅ WiFive Pro Studio rodando em http://localhost:${PORT}`);
  console.log(`   (Compatível com qualquer versão do Linux Mint e Node.js)\n`);
});

# Relatório de Engenharia e Documentação Técnica: WiFive vs. WiFive Pro Studio

## 1. Visão Geral e Análise do Sistema Original (`painel-skills.vercel.app/wifive/`)

O aplicativo original **WiFive Bordas** foi concebido como uma ferramenta web para criadores de canais dark, clipmakers de podcast, páginas de fofoca, futebol e notícias curtas. Seu objetivo primordial era reduzir o tempo de edição manual ao aplicar uma mesma moldura institucional (com nome, @ e logotipo) sobre múltiplos vídeos simultaneamente.

### Como o WiFive original funcionava:
1. **Frontend**: Monólito em HTML com scripts embutidos (`index.html` de ~265 KB contendo login, guias e wrapper do painel de skills).
2. **Entrada de Dados**:
   - Campos de texto simples: Nome da página, `@` de rede social, texto de chamada no topo.
   - Upload de imagem de perfil (logo) desenhada em canvas 2D.
   - 5 estilos estáticos: *Post estilo X*, *Neon*, *Notícia Urgente*, *Gradiente* e *Meme (Impacto)*, além de suporte a imagem PNG transparente externa.
   - Seleção múltipla de arquivos de vídeo locais via `<input type="file" multiple>`.
3. **Pipeline de Renderização Original**:
   - Um elemento `<canvas>` estático em memória com tamanho pré-fixado em 720×1280.
   - Um elemento `<video>` oculto carregava cada arquivo via `URL.createObjectURL`.
   - Um loop `requestAnimationFrame` desenhava o frame atual do vídeo no canvas e sobrepunha o cabeçalho/moldura.
   - O canvas chamava `canvas.captureStream()` e gravava com `MediaRecorder`.
   - Ao finalizar, gerava um Blob e disparava um download com extensão `.mp4`.

---

## 2. Diagnóstico de Gargalos de Performance e Instabilidade do WiFive Original

| Ponto Crítico | Comportamento no WiFive Original | Consequência para o Usuário |
| :--- | :--- | :--- |
| **Falso MP4 (WebM Container)** | Usava `MediaRecorder` sem validar suporte de codec h264/avc1. Gravava WebM/VP8 padrão e simplesmente renomeava a extensão para `.mp4`. | Erro ao salvar na galeria do iPhone (iOS fotos não abre), falha de upload no Instagram Reels e TikTok ("formato não suportado"). |
| **Resoluções Baixas e Rígidas** | Limitado a 720p ou resolução arbitrária do vídeo original sem upscaling bicúbico ou controle de bitrate. | Vídeos ficavam pixelados ou borrados em telas modernas Full HD / 2K, diminuindo o alcance nas redes. |
| **Vazamento de Memória (Memory Leak)** | Não limpava `URL.createObjectURL`, contexts de áudio nem instâncias de `MediaRecorder` no lote. | Ao processar mais de 10 vídeos no celular, a aba travava silenciosamente por esgotamento de RAM (OOM Crash). |
| **Perda de Sincronia de Áudio** | Capturava áudio apenas do `captureStream()` sem nó de ganho do `AudioContext`. | Vídeos exportados com cortes de áudio nos primeiros 500ms ou dessincronia labial progressiva. |
| **Sem Edição ou Corte de Início/Fim** | Obrigava o usuário a cortar o clipe antes em outros apps (CapCut). | Fluxo de trabalho quebrado e perda de horas. |
| **Sem Recorte de Bordas (Crop)** | Não permitia remover faixas pretas, legendas antigas ou marcas d'água existentes nos clipes. | Vídeos ficavam sujos com barras pretas duplas na moldura. |
| **Penalização por Algoritmo (Shadowban)** | Mantinha ou truncava os metadados brutos do container, permitindo que as redes identificassem clipes duplicados. | Perda de alcance e detecção de "conteúdo reutilizado". |

---

## 3. Auditoria de Segurança e Vulnerabilidades do Painel Original

1. **Autenticação Insegura**:
   - O painel efetuava chamadas para `/api/conta` transmitindo chaves derivadas em texto sem rotação ou HMAC validado contra CSRF.
2. **Exposição de Dados Sensíveis**:
   - Armazenava tokens de identificação de máquina (`maquina()`) e e-mail em `localStorage` sem hash criptográfico forte.
3. **Injeção de Código em Canvas**:
   - Nomes e `@` longos eram renderizados diretamente sem sanitização de caracteres especiais ou cálculo dinâmico de quebra de linha, ocasionando overflow visual ou bugs no layout.

---

## 4. Arquitetura e Inovações do **WiFive Pro Studio**

O **WiFive Pro Studio** foi projetado do zero com arquitetura moderna em React 19, TypeScript e Tailwind CSS, trazendo um salto qualitativo profissional:

### A. Estabilidade Absoluta de Download & Suporte Real a MP4 / Alta Definição
- **Detecção Inteligente de Codecs**: Prioriza `video/mp4;codecs=avc1.42E01E,mp4a.40.2` e codecs H.264 nativos com fallback seguro.
- **Resoluções em Alta Definição Selecionáveis**:
  - **720p HD** (720×1280 em 9:16)
  - **1080p Full HD** (1080×1920 em 9:16 / 1080×1080 em 1:1) — Nitidez cristalina ideal para TikTok/Reels/Shorts
  - **1440p 2K Ultra** (1440×2560 em 9:16) — Para monitores e criadores de alta fidelidade
- **Bitrate Configurável**: Até 16 Mbps para preservar detalhes finos em cenas de alta movimentação.
- **Download em Lote em Arquivo ZIP Único**: Integrado com `JSZip`, permitindo baixar 50 vídeos com 1 único clique.

### B. Ferramenta de Corte Visual de Timeline (Video Trimmer)
- Seleção milimétrica de ponto de início (`startTime`) e fim (`endTime`).
- Calculadora de duração real resultante.
- Prevenção de renderização desnecessária, economizando bateria e tempo.

### C. Recorte de Bordas Interativo (Edge Cropper) & Ajuste de Escala
- **Remoção de Bordas nos 4 Lados**: Sliders para recortar topo, base, esquerda e direita (elimina marcas d'água antigas do TikTok, legendas indesejadas e barras pretas).
- **Zoom & Escala (0.5x a 2.0x)**: Ajuste fino do tamanho do vídeo dentro da área útil da moldura.
- **Posicionamento X e Y**: Permite enquadrar o rosto do orador no centro perfeito.
- **Modos de Enquadramento**:
  - *Conter com Fundo Desfocado Dinâmico*: Renderiza o próprio vídeo com blur em tempo real no fundo para preencher espaço.
  - *Conter com Fundo Sólido*: Estilo minimalista escuro.
  - *Preencher (Cover)*: Preenche 100% da área útil.

### D. Sanitização Total de Metadados (Anti-Shadowban / Algoritmo Limpo)
- Ao transcodificar os frames do vídeo e amostras de áudio através do pipeline WebCodecs/Canvas/WebAudio, todos os metadados de câmeras (EXIF, GPS, timestamps do dispositivo original, assinaturas de apps como CapCut/InShot) são completamente expurgados.
- O arquivo MP4 resultante é gerado como mídia sintética virgem, prevenindo flags automatizadas de reutilização de conteúdo.

### E. Efeitos Visuais & Correção de Cor em Tempo Real
- Controle de **Brilho** (0.5x a 1.5x)
- Controle de **Contraste** (0.5x a 1.5x)
- Controle de **Saturação** (Preto & Branco 0x até Super Punchy 2x)
- Filtro de **Nitidez e Vinheta Cinematográfica**
- Desfoque de Fundo ajustável de 0 a 40 pixels

### F. Controle de Velocidade com Preservação de Tom
- Modulação de velocidade: 0.5x, 0.75x, 1.0x, **1.1x (ritmo viral ideal)**, 1.25x, 1.5x e 2.0x.
- Mantém o tom da voz natural sem efeito de "esquilo" (`preservesPitch = true`).

### G. Áudio de Fundo (BGM) & Mixagem com Controle de Volume
- Sistema de áudio com Web Audio API multicanal:
  - Controle de volume da voz/vídeo original (0% a 200%).
  - Controle de volume da trilha de fundo (0% a 100%).
  - 5 trilhas sonoras integradas livres de direitos (Lo-Fi, Brazilian Phonk, Plantão Suspense, Trap Viral, Podcast Chill).
  - Suporte a upload de qualquer arquivo MP3/WAV/M4A do usuário.
  - Fade-out automático no final do clipe.

### H. Variedade de Molduras Modernas
- 8 estilos pré-configurados prontos para uso + suporte a moldura PNG personalizada + modo sem moldura (apenas cortes e efeitos).
- Design 100% responsivo para celulares, tablets e computadores, com interface escura moderna e acessível.

---

## 5. Garantia de Privacidade e Processamento 100% Local (Client-Side)
Nenhum byte de vídeo do usuário é transferido para servidores de terceiros ou nuvem. Todo o processamento ocorre no próprio hardware do usuário através das APIs nativas do navegador (Web Audio, Canvas 2D acelerado por hardware e MediaRecorder), garantindo sigilo absoluto, ausência de custos de infraestrutura e velocidade máxima sem filas.

        JSON.stringify({
          pageName: brand.pageName,
          handle: brand.handle,
          badgeType: brand.badgeType,
          accentColor: brand.accentColor,
        })
      );
    } catch (_) {}
  }, [brand.pageName, brand.handle, brand.badgeType, brand.accentColor]);

  // Selected Active Video Item (memoized to keep reference stable)
  const activeItem = React.useMemo(() => {
    return queue.find((item) => item.id === activeVideoId) || queue[0] || null;
  }, [queue, activeVideoId]);

  // Add Files to Queue
  const handleAddFiles = (files: FileList | File[]) => {
    const fileArray = Array.from(files);
    const newItems: VideoQueueItem[] = fileArray.map((file) => {
      const url = URL.createObjectURL(file);
      return {
        id: `vid_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`,
        file,
        url,
        name: file.name,
        duration: 0,
        width: 0,
        height: 0,
        thumbnailUrl: '',
        trim: { startTime: 0, endTime: 0 },
        status: 'idle',
        progress: 0,
        outputBlob: null,
        outputUrl: null,
        errorMessage: null,
      };
    });

    setQueue((prev) => [...prev, ...newItems]);
    if (!activeVideoId && newItems.length > 0) {
      setActiveVideoId(newItems[0].id);
    }

    // Load metadata and thumbnails in background
    newItems.forEach((item) => {
      const tempVideo = document.createElement('video');
      tempVideo.crossOrigin = 'anonymous';
      tempVideo.preload = 'metadata';
      tempVideo.src = item.url;
      tempVideo.onloadedmetadata = () => {
        const duration = tempVideo.duration || 10;
        const width = tempVideo.videoWidth || 720;
        const height = tempVideo.videoHeight || 1280;

        // capture quick thumbnail at 1s or 0s
        tempVideo.currentTime = Math.min(1, duration * 0.1);
        tempVideo.onseeked = () => {
          const thumbCanvas = document.createElement('canvas');
          thumbCanvas.width = 120;
          thumbCanvas.height = Math.round((120 / width) * height);
          const tCtx = thumbCanvas.getContext('2d');
          if (tCtx) {
            tCtx.drawImage(tempVideo, 0, 0, thumbCanvas.width, thumbCanvas.height);
            const thumbUrl = thumbCanvas.toDataURL('image/jpeg');
            setQueue((q) =>
              q.map((v) =>
                v.id === item.id
                  ? {
                      ...v,
                      duration,
                      width,
                      height,
                      thumbnailUrl: thumbUrl,
                      trim: { startTime: 0, endTime: duration },
                    }
                  : v
              )
            );
          }
        };
      };
    });
  };

  const handleRemoveVideo = (id: string) => {
    setQueue((prev) => {
      const filtered = prev.filter((item) => item.id !== id);
      if (activeVideoId === id) {
        setActiveVideoId(filtered.length > 0 ? filtered[0].id : null);
      }
      return filtered;
    });
  };

  const handleLoadDemoSample = async () => {
    const demo = await createSyntheticDemoVideo();
    setQueue((prev) => [...prev, demo]);
    setActiveVideoId(demo.id);
  };

  // Trimming update for active video
  const handleUpdateTrim = (startTime: number, endTime: number) => {
    if (!activeItem) return;
    setQueue((prev) =>
      prev.map((item) =>
        item.id === activeItem.id ? { ...item, trim: { startTime, endTime } } : item
      )
    );
  };

  // Individual headline update per video
  const handleUpdateItemHeadline = (id: string, headline: string) => {
    setQueue((prev) =>
      prev.map((item) => (item.id === id ? { ...item, customHeadline: headline } : item))
    );
  };

  // Batch headlines by lines
  const handleBatchSetHeadlines = (headlines: string[]) => {
    setQueue((prev) =>
      prev.map((item, idx) => {
        if (idx < headlines.length) {
          return { ...item, customHeadline: headlines[idx] };
        }
        return item;
      })
    );
  };

  // Batch Processing
  const handleProcessAll = async () => {
    if (queue.length === 0 || isProcessing) return;

    setIsProcessing(true);
    abortControllerRef.current = new AbortController();

    try {
      for (let i = 0; i < queue.length; i++) {
        const item = queue[i];
        if (abortControllerRef.current?.signal.aborted) break;

        // Set active for preview
        setActiveVideoId(item.id);

        // Update item status
        setQueue((q) =>
          q.map((v) => (v.id === item.id ? { ...v, status: 'processing', progress: 5 } : v))
        );

        try {
          const result = await processVideoItem(
            item,
            frameStyle,
            brand,
            effects,
            audioSettings,
            cropSettings,
            exportSettings,
            (progress, statusText) => {
              setQueue((q) =>
                q.map((v) => (v.id === item.id ? { ...v, progress } : v))
              );
            },
            abortControllerRef.current?.signal
          );

          setQueue((q) =>
            q.map((v) =>
              v.id === item.id
                ? {
                    ...v,
                    status: 'done',
                    progress: 100,
                    outputBlob: result.outputBlob,
                    outputUrl: result.outputUrl,
                    errorMessage: null,
                  }
                : v
            )
          );
        } catch (err: any) {
          if (abortControllerRef.current?.signal.aborted) break;
          setQueue((q) =>
            q.map((v) =>
              v.id === item.id
                ? {
                    ...v,
                    status: 'error',
                    progress: 0,
                    errorMessage: err.message || 'Erro no processamento',
                  }
                : v
            )
          );
        }
      }
    } finally {
      setIsProcessing(false);
      abortControllerRef.current = null;
    }
  };

  const handleCancelProcess = () => {
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
    }
    setIsProcessing(false);
  };

  const handleDownloadSingle = (item: VideoQueueItem) => {
    downloadSingleVideo(item);
  };

  const handleDownloadZip = () => {
    downloadAllVideosAsZip(queue);
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans selection:bg-blue-500 selection:text-white">
      {/* App Header */}
      <Header
        exportSettings={exportSettings}
        onUpdateExportSettings={(updated) => setExportSettings((prev) => ({ ...prev, ...updated }))}
        onOpenDocs={() => setIsDocsOpen(true)}
        queueCount={queue.length}
      />

      {/* Main Content Area */}
      <main className="flex-1 max-w-7xl w-full mx-auto p-3 sm:p-5 lg:p-6 grid grid-cols-1 lg:grid-cols-12 gap-6">
        {/* Left Column: Interactive Studio Controls (Tabs / Panels) */}
        <div className="lg:col-span-7 flex flex-col space-y-4">
          {/* Studio Navigation Tabs */}
          <div className="flex bg-slate-900 border border-slate-800 rounded-xl p-1 gap-1 overflow-x-auto text-xs font-semibold">
            <button
              type="button"
              onClick={() => setActiveTab('frame')}
              className={`flex-1 py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition whitespace-nowrap ${
                activeTab === 'frame'
                  ? 'bg-blue-600 text-white shadow'
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-850'
              }`}
            >
              <Sparkles className="w-3.5 h-3.5 text-cyan-300" />
              Moldura & Marca
            </button>

            <button
              type="button"
              onClick={() => setActiveTab('crop')}
              className={`flex-1 py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition whitespace-nowrap ${
                activeTab === 'crop'
                  ? 'bg-blue-600 text-white shadow'
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-850'
              }`}
            >
              <Scissors className="w-3.5 h-3.5 text-amber-300" />
              Corte & Bordas
            </button>

            <button
              type="button"
              onClick={() => setActiveTab('effects')}
              className={`flex-1 py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition whitespace-nowrap ${
                activeTab === 'effects'
                  ? 'bg-blue-600 text-white shadow'
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-850'
              }`}
            >
              <Sliders className="w-3.5 h-3.5 text-fuchsia-300" />
              Efeitos & Áudio
            </button>

            <button
              type="button"
              onClick={() => setActiveTab('queue')}
              className={`flex-1 py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition whitespace-nowrap ${
                activeTab === 'queue'
                  ? 'bg-blue-600 text-white shadow'
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-850'
              }`}
            >
              <Film className="w-3.5 h-3.5 text-emerald-300" />
              Fila em Lote ({queue.length})
            </button>
          </div>

          {/* Active Tab Panel */}
          <div className="space-y-4">
            {activeTab === 'frame' && (
              <FrameEditor
                frameStyle={frameStyle}
                onSelectFrameStyle={setFrameStyle}
                brand={brand}
                onUpdateBrand={(up) => setBrand((prev) => ({ ...prev, ...up }))}
              />
            )}

            {activeTab === 'crop' && (
              <VideoTrimmerCrop
                activeItem={activeItem}
                cropSettings={cropSettings}
                onUpdateCrop={(up) => setCropSettings((prev) => ({ ...prev, ...up }))}
                onUpdateTrim={handleUpdateTrim}
              />
            )}

            {activeTab === 'effects' && (
              <EffectsAudioPanel
                effects={effects}
                onUpdateEffects={(up) => setEffects((prev) => ({ ...prev, ...up }))}
                audioSettings={audioSettings}
                onUpdateAudio={(up) => setAudioSettings((prev) => ({ ...prev, ...up }))}
                exportSettings={exportSettings}
                onUpdateExportSettings={(up) => setExportSettings((prev) => ({ ...prev, ...up }))}
              />
            )}

            {activeTab === 'queue' && (
              <BatchQueue
                queue={queue}
                activeId={activeVideoId}
                onSelectVideo={setActiveVideoId}
                onAddFiles={handleAddFiles}
                onRemoveVideo={handleRemoveVideo}
                onProcessAll={handleProcessAll}
                onCancelProcess={handleCancelProcess}
                isProcessing={isProcessing}
                onDownloadSingle={handleDownloadSingle}
                onDownloadZip={handleDownloadZip}
                onLoadDemoSample={handleLoadDemoSample}
                onUpdateItemHeadline={handleUpdateItemHeadline}
                onBatchSetHeadlines={handleBatchSetHeadlines}
              />
            )}
          </div>

          {/* Bottom helper card if not on queue tab */}
          {activeTab !== 'queue' && (
            <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-3.5 flex items-center justify-between gap-3 text-xs">
              <div className="flex items-center gap-2">
                <Film className="w-4 h-4 text-blue-400 flex-shrink-0" />
                <span className="text-slate-300">
                  {queue.length} {queue.length === 1 ? 'vídeo na fila' : 'vídeos na fila de processamento'}
                </span>
              </div>
              <button
                type="button"
                onClick={() => setActiveTab('queue')}
                className="text-blue-400 hover:text-blue-300 font-bold transition flex items-center gap-1"
              >
                Gerenciar Fila & Baixar ›
              </button>
            </div>
          )}
        </div>

        {/* Right Column: Live 60 FPS Preview & Batch Action */}
        <div className="lg:col-span-5 space-y-4">
          <div className="sticky top-20 space-y-4">
            <LiveCanvasPreview
              activeItem={activeItem}
              frameStyle={frameStyle}
              brand={brand}
              effects={effects}
              cropSettings={cropSettings}
              exportSettings={exportSettings}
              onDownloadActive={() => activeItem && handleDownloadSingle(activeItem)}
              isProcessing={isProcessing}
            />

            {/* Quick Process CTA directly below preview */}
            {queue.length > 0 && activeTab !== 'queue' && (
              <button
                type="button"
                onClick={handleProcessAll}
                disabled={isProcessing}
                className="w-full py-3.5 px-4 rounded-xl bg-gradient-to-r from-blue-600 via-indigo-600 to-cyan-500 hover:from-blue-500 hover:to-cyan-400 text-white font-extrabold text-sm flex items-center justify-center gap-2 shadow-xl shadow-blue-500/20 active:scale-95 transition disabled:opacity-50"
              >
                <Sparkles className="w-4 h-4" />
                {queue.length === 1
                  ? 'Processar & Gerar MP4 HD'
                  : `Processar Fila Inteira (${queue.length} vídeos)`}
              </button>
            )}
          </div>
        </div>
      </main>

      {/* Technical Documentation Modal */}
      <DocumentationModal isOpen={isDocsOpen} onClose={() => setIsDocsOpen(false)} />
    </div>
  );
}

  const completedCount = queue.filter((i) => i.status === 'done').length;

  return (

            </h2>
            <p className="text-[11px] text-slate-400">
              Arraste múltiplos cortes para aplicar a moldura em todos
            </p>
          </div>
        </div>

        {/* Action Buttons in Header */}
        <div className="flex items-center gap-2">
          {queue.length > 0 && (
            <button
              type="button"
              onClick={() => {
                // Pre-fill existing headlines into textarea
                const existing = queue.map((v) => v.customHeadline || '').join('\n');
                setLinesText(existing);
                setShowBatchTitlesModal(true);
              }}
              className="px-2.5 py-1.5 rounded-lg bg-blue-600/20 hover:bg-blue-600/30 text-blue-300 border border-blue-500/30 text-xs font-semibold flex items-center gap-1.5 transition"
            >
              <ListOrdered className="w-3.5 h-3.5 text-blue-400" />
              Títulos por Linha

          )}

          {/* Demo Button if empty */}
          {queue.length === 0 && (
            <button

              onClick={onLoadDemoSample}
              className="px-2.5 py-1.5 rounded-lg bg-indigo-600/20 hover:bg-indigo-600/30 text-indigo-300 border border-indigo-500/30 text-xs font-semibold flex items-center gap-1.5 transition"
            >
              <Sparkles className="w-3.5 h-3.5 text-indigo-400" />
              Carregar Vídeo de Exemplo
            </button>
          )}
        </div>
      </div>

      {/* Banner de download do programa para Linux Mint */}
      <div className="p-3 rounded-xl bg-gradient-to-r from-emerald-950/40 via-slate-900 to-teal-950/40 border border-emerald-500/30 flex items-center justify-between gap-3 flex-wrap">
        <div className="flex items-center gap-2.5">

            <Download className="w-4 h-4" />
          </div>
          <div>
            <div className="text-xs font-bold text-white flex items-center gap-2">

            </p>
          </div>
        </div>
        <button
          type="button"
          onClick={() => downloadLinuxPackage()}
          className="px-3 py-1.5 rounded-lg bg-emerald-600 hover:bg-emerald-500 text-white font-bold text-xs flex items-center gap-1.5 shadow-md transition active:scale-95 cursor-pointer ml-auto"
        >
          <Download className="w-3.5 h-3.5" />
          Baixar Arquivo Linux (.zip)
        </button>
      </div>

      {/* Upload Dropzone */}
      <input
        ref={fileInputRef}
        type="file"
        multiple
        accept="video/mp4,video/quicktime,video/webm,video/x-matroska"
        onChange={handleFileChange}

      />

      <div
        onDragOver={(e) => e.preventDefault()}
        onDrop={handleDrop}
        onClick={() => fileInputRef.current?.click()}
        className="border-2 border-dashed border-slate-700/80 hover:border-blue-500/80 bg-slate-950/50 hover:bg-slate-950/80 rounded-xl p-5 text-center cursor-pointer transition group"
      >
        <div className="w-10 h-10 rounded-full bg-blue-500/10 border border-blue-500/20 flex items-center justify-center text-blue-400 mx-auto mb-2 group-hover:scale-110 transition">
          <Upload className="w-5 h-5" />
        </div>
        <p className="text-xs font-bold text-white">
          Toque para escolher os vídeos ou arraste a pasta inteira
        </p>
        <p className="text-[10px] text-slate-400 mt-1">
          Suporta MP4, MOV, WebM em lote · Processamento 100% no seu dispositivo
        </p>
      </div>

      {/* Video List */}
      {queue.length > 0 && (
        <div className="space-y-2.5 max-h-[380px] overflow-y-auto pr-1">
          {queue.map((item, idx) => {
            const isSelected = item.id === activeId;
            return (
              <div
                key={item.id}
                onClick={() => onSelectVideo(item.id)}
                className={`p-3 rounded-xl border flex flex-col gap-2 cursor-pointer transition ${
                  isSelected
                    ? 'bg-blue-600/15 border-blue-500/70 ring-1 ring-blue-500/30 text-white'
                    : 'bg-slate-950/60 border-slate-800 text-slate-300 hover:border-slate-700 hover:bg-slate-850'
                }`}
              >
                <div className="flex items-center justify-between gap-3">
                  {/* Thumbnail & Info */}
                  <div className="flex items-center gap-3 min-w-0">
                    <div className="relative w-12 h-14 rounded-lg bg-slate-900 border border-slate-800 flex-shrink-0 overflow-hidden flex items-center justify-center">
                      {item.thumbnailUrl ? (
                        <img
                          src={item.thumbnailUrl}
                          alt="Preview"
                          className="w-full h-full object-cover"
                        />
                      ) : (
                        <Film className="w-5 h-5 text-slate-600" />
                      )}
                      <span className="absolute bottom-0.5 right-0.5 text-[8px] bg-black/80 px-1 rounded text-slate-300 font-mono">
                        {formatTime(item.duration)}
                      </span>
                    </div>

                    <div className="min-w-0">
                      <div className="flex items-center gap-2">
                        <span className="text-[10px] font-bold text-slate-400">#{idx + 1}</span>
                        <h4 className="text-xs font-bold text-white truncate max-w-[180px] sm:max-w-[260px]">
                          {item.name}
                        </h4>
                      </div>
                      <div className="text-[10px] text-slate-400 flex items-center gap-2 mt-0.5">
                        <span>Corte: {item.trim.startTime.toFixed(1)}s - {item.trim.endTime.toFixed(1)}s</span>
                        {item.width > 0 && <span>({item.width}×{item.height})</span>}
                      </div>

                      {/* Progress bar if processing */}
                      {item.status === 'processing' && (
                        <div className="w-full bg-slate-800 rounded-full h-1.5 mt-1.5 overflow-hidden">
                          <div
                            className="bg-blue-500 h-full transition-all duration-200"
                            style={{ width: `${item.progress}%` }}
                          />
                        </div>
                      )}
                    </div>
                  </div>

                  {/* Status & Actions */}
                  <div className="flex items-center gap-2 flex-shrink-0" onClick={(e) => e.stopPropagation()}>
                    {item.status === 'idle' && (
                      <span className="text-[10px] px-2 py-0.5 rounded bg-slate-800 text-slate-400 border border-slate-700">
                        Pronto
                      </span>
                    )}

                    {item.status === 'processing' && (
                      <span className="text-[10px] px-2 py-0.5 rounded bg-blue-500/20 text-blue-300 border border-blue-500/30 flex items-center gap-1">
                        <Loader2 className="w-3 h-3 animate-spin" />
                        {item.progress}%
                      </span>
                    )}

                    {item.status === 'done' && (
                      <div className="flex items-center gap-1.5">
                        <button
                          type="button"
                          onClick={() => onDownloadSingle(item)}
                          title="Baixar vídeo gerado"
                          className="px-2.5 py-1 rounded-lg bg-emerald-600 hover:bg-emerald-500 text-white text-[11px] font-bold flex items-center gap-1 transition shadow-sm"
                        >
                          <Download className="w-3 h-3" />
                          {item.outputBlob?.type.includes('mp4') ? 'MP4' : 'Baixar HD'}
                        </button>
                      </div>
                    )}

                    {item.status === 'error' && (
                      <span className="text-[10px] px-2 py-0.5 rounded bg-red-500/20 text-red-300 border border-red-500/30 flex items-center gap-1">
                        <AlertCircle className="w-3 h-3" /> Erro
                      </span>
                    )}

                    <button
                      type="button"
                      onClick={() => onRemoveVideo(item.id)}
                      title="Remover vídeo da fila"

                type="button"
                onClick={onProcessAll}
                className="flex-1 sm:flex-initial px-5 py-2.5 rounded-xl bg-gradient-to-r from-blue-600 to-cyan-500 hover:from-blue-500 hover:to-cyan-400 text-white text-xs font-extrabold flex items-center justify-center gap-2 transition shadow-lg shadow-blue-500/25 active:scale-95"
              >
                <Sparkles className="w-4 h-4" />
                {queue.length === 1 ? 'Gerar Vídeo MP4' : `Aplicar em Todos (${queue.length})`}
              </button>
            )}
          </div>
        </div>
      )}
    </div>
  );
};

import React, { useEffect, useRef, useState } from 'react';
import {
  FrameStyle,

  VisualEffects,
  VideoCropSettings,
  ExportSettings,

} from '../types/video';
import { calculateSourceCropRect, calculateDestPlacement, formatTime } from '../utils/geometryMath';
import { getFrameLayoutSlot, renderBackground, renderFrameOverlay } from '../utils/frameRenderer';
import {
  Play,
  Pause,
  Maximize2,
  Download,
  RotateCcw,
  Sparkles,
  Smartphone,
  Sliders,
} from 'lucide-react';

interface LiveCanvasPreviewProps {
  activeItem: VideoQueueItem | null;
  frameStyle: FrameStyle;
  brand: ProfileBrand;
  effects: VisualEffects;
  cropSettings: VideoCropSettings;
  exportSettings: ExportSettings;
  onDownloadActive?: () => void;
  isProcessing?: boolean;
}

export const LiveCanvasPreview: React.FC<LiveCanvasPreviewProps> = ({
  activeItem,
  frameStyle,
  brand,
  effects,
  cropSettings,
  exportSettings,
  onDownloadActive,
  isProcessing = false,
}) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const videoRef = useRef<HTMLVideoElement>(null);
  const timeDisplayRef = useRef<HTMLElement>(null);
  const [isPlaying, setIsPlaying] = useState<boolean>(true);

  // Hidden image elements for avatar and custom frame
  const avatarImgRef = useRef<HTMLImageElement | null>(null);
  const customFrameImgRef = useRef<HTMLImageElement | null>(null);

  // Extract stable primitives from activeItem to avoid effect re-runs when progress/status updates
  const activeItemId = activeItem?.id;
  const activeItemUrl = activeItem?.url;
  const trimStart = activeItem?.trim.startTime || 0;
  const trimEnd = activeItem?.trim.endTime || (activeItem?.duration ?? 10);
  const customHeadline = activeItem?.customHeadline;

  useEffect(() => {
    if (brand.avatarUrl) {
      const img = new Image();
      img.crossOrigin = 'anonymous';
      img.src = brand.avatarUrl;
      avatarImgRef.current = img;
    } else {
      avatarImgRef.current = null;
    }
  }, [brand.avatarUrl]);

  useEffect(() => {
    if (brand.customFrameUrl && frameStyle === 'custom') {
      const img = new Image();
      img.crossOrigin = 'anonymous';
      img.src = brand.customFrameUrl;
      customFrameImgRef.current = img;
    } else {
      customFrameImgRef.current = null;
    }
  }, [brand.customFrameUrl, frameStyle]);

  // Handle video element setup only when video source changes
  useEffect(() => {
    const video = videoRef.current;
    if (!video || !activeItemUrl) return;

    video.src = activeItemUrl;
    video.currentTime = trimStart;
    video.playbackRate = exportSettings.speed || 1.0;

    const playPromise = video.play();
    if (playPromise !== undefined) {
      playPromise
        .then(() => setIsPlaying(true))
        .catch(() => setIsPlaying(false));
    }
  }, [activeItemId, activeItemUrl]);

  // Adjust playback rate when speed setting changes without reloading video
  useEffect(() => {
    if (videoRef.current) {
      videoRef.current.playbackRate = exportSettings.speed || 1.0;
    }
  }, [exportSettings.speed]);

  // Main rendering loop - zero setState calls inside requestAnimationFrame
  useEffect(() => {
    if (isProcessing) {
      if (videoRef.current && !videoRef.current.paused) {
        videoRef.current.pause();
      }
      return;
    }

    let animId: number;

    const hasCustomFilters =
      effects.brightness !== 1 ||
      effects.contrast !== 1 ||
      effects.saturation !== 1 ||
      effects.sharpen;

    let filterStr = '';
    if (hasCustomFilters) {
      filterStr = `brightness(${effects.brightness}) contrast(${effects.contrast}) saturate(${effects.saturation})`;
      if (effects.sharpen) {
        filterStr += ' contrast(1.15)';
      }
    }

    const render = () => {
      const canvas = canvasRef.current;
      const video = videoRef.current;
      if (!canvas) return;

      const ctx = canvas.getContext('2d');
      if (!ctx) return;

      const w = canvas.width;
      const h = canvas.height;

      // Handle video loop inside trim bounds and update time DOM ref directly
      if (video && activeItemUrl) {
        const start = trimStart;
        const end = trimEnd > start ? trimEnd : video.duration || 10;

        if (video.currentTime >= end || video.currentTime < start) {
          video.currentTime = start;
        }
        if (timeDisplayRef.current) {
          timeDisplayRef.current.textContent = formatTime(video.currentTime);
        }
      }

      // 1. Render Background
      renderBackground(
        ctx,
        w,
        h,
        video,
        effects,
        cropSettings.fitMode,
        cropSettings.backdropColor
      );

      // 2. Render Video
      if (video && video.readyState >= 2) {
        const slot = getFrameLayoutSlot(frameStyle);
        const srcRect = calculateSourceCropRect(video.videoWidth, video.videoHeight, cropSettings);
        const croppedAspect = srcRect.sw / srcRect.sh;
        const destRect = calculateDestPlacement(
          w,
          h,
          croppedAspect,
          slot,
          cropSettings.scale,
          cropSettings.positionX,
          cropSettings.positionY
        );

        ctx.save();
        if (hasCustomFilters) {
          ctx.filter = filterStr;
        }

        ctx.drawImage(
          video,
          srcRect.sx,
          srcRect.sy,
          srcRect.sw,
          srcRect.sh,
          destRect.dx,
          destRect.dy,
          destRect.dw,
          destRect.dh
        );
        ctx.restore();

        // Optional vignette
        if (effects.vignette) {
          ctx.save();
          const vigGrad = ctx.createRadialGradient(
            w / 2,
            h / 2,
            w * 0.35,
            w / 2,
            h / 2,
            w * 0.75
          );
          vigGrad.addColorStop(0, 'rgba(0,0,0,0)');
          vigGrad.addColorStop(1, 'rgba(0,0,0,0.55)');
          ctx.fillStyle = vigGrad;
          ctx.fillRect(0, 0, w, h);
          ctx.restore();
        }
      }

      // 3. Render Frame Overlay
      renderFrameOverlay(
        ctx,
        w,
        h,
        frameStyle,
        brand,
        avatarImgRef.current,
        customFrameImgRef.current,
        video ? video.currentTime : 0,
        customHeadline
      );

      // Only schedule next frame if actively playing to eliminate idle CPU drain
      if (isPlaying && video && !video.paused) {
        animId = requestAnimationFrame(render);
      }
    };

    render();

    return () => {
      if (animId) cancelAnimationFrame(animId);
    };
  }, [
    isProcessing,
    isPlaying,
    activeItemId,
    activeItemUrl,
    trimStart,
    trimEnd,
    customHeadline,
    frameStyle,
    brand,
    effects,
    cropSettings,
    exportSettings.aspectRatio,
  ]);

  const togglePlay = () => {
    const video = videoRef.current;
    if (!video) return;
    if (video.paused) {
      video.play();
      setIsPlaying(true);
    } else {
      video.pause();
      setIsPlaying(false);
    }
  };

  const restartClip = () => {
    const video = videoRef.current;
    if (!video) return;
    video.currentTime = trimStart;
    if (timeDisplayRef.current) {
      timeDisplayRef.current.textContent = formatTime(video.currentTime);
    }
    video.play();
    setIsPlaying(true);
  };

  const aspectClass =
    exportSettings.aspectRatio === '9:16'
      ? 'aspect-[9/16] max-h-[580px]'
      : exportSettings.aspectRatio === '1:1'
      ? 'aspect-square max-h-[520px]'
      : 'aspect-video max-h-[380px]';

  return (
    <div className="bg-slate-900/90 border border-slate-800 rounded-2xl p-4 sm:p-5 shadow-lg flex flex-col items-center">
      {/* Top Header */}
      <div className="w-full flex items-center justify-between pb-3 border-b border-slate-800 mb-4">
        <div className="flex items-center gap-2">
          <div className="w-2.5 h-2.5 rounded-full bg-emerald-500 animate-pulse" />
          <h3 className="text-sm font-bold text-white">Prévia em Tempo Real (60 FPS)</h3>
        </div>
        <div className="flex items-center gap-2 text-xs">
          <span className="px-2 py-0.5 rounded bg-slate-800 text-slate-300 font-mono font-semibold border border-slate-700">
            {exportSettings.resolution} · {exportSettings.aspectRatio}
          </span>
        </div>
      </div>

      {/* Hidden Native Video for Canvas playback */}
      <video
        ref={videoRef}
        playsInline
        muted
        loop
        crossOrigin="anonymous"
        className="hidden"
      />

      {/* Canvas Display Container (Phone Mockup Styling) */}
      <div className="relative group w-full flex items-center justify-center p-2 bg-slate-950/80 rounded-2xl border border-slate-800/80 shadow-2xl overflow-hidden">
        <div className={`relative ${aspectClass} w-auto shadow-2xl rounded-xl overflow-hidden border border-slate-800 bg-black`}>
          <canvas
            ref={canvasRef}
            width={720}
            height={exportSettings.aspectRatio === '9:16' ? 1280 : exportSettings.aspectRatio === '1:1' ? 720 : 405}
            className="w-full h-full object-contain cursor-pointer"
            onClick={togglePlay}
          />

          {/* Quick Play/Pause Center Overlay on Hover */}
          {!isProcessing && (
            <button
              type="button"
              onClick={togglePlay}
              aria-label={isPlaying ? 'Pausar prévia' : 'Reproduzir prévia'}
              className="absolute inset-0 flex items-center justify-center bg-black/30 opacity-0 group-hover:opacity-100 transition-opacity"
            >
              <div className="w-12 h-12 rounded-full bg-slate-900/90 border border-slate-600 text-white flex items-center justify-center shadow-lg">
                {isPlaying ? <Pause className="w-5 h-5" /> : <Play className="w-5 h-5 ml-0.5" />}
              </div>
            </button>
          )}

          {/* Processing Overlay to free up 100% GPU/CPU */}
          {isProcessing && (
            <div className="absolute inset-0 bg-slate-950/90 backdrop-blur-sm flex flex-col items-center justify-center p-5 text-center z-20">
              <div className="w-11 h-11 rounded-xl bg-blue-600/20 border border-blue-500/40 text-blue-400 flex items-center justify-center mb-3">
                <Sparkles className="w-5 h-5 animate-spin" style={{ animationDuration: '3s' }} />
              </div>
              <span className="text-xs font-bold text-white mb-1">Renderizando em 1080p HD</span>
              <span className="text-[11px] text-slate-400 max-w-[210px] leading-relaxed">
                Prévia pausada para alocar máxima aceleração de hardware sem travamentos.
              </span>
            </div>
          )}

          {/* Empty State when no video is selected */}
          {!activeItem && !isProcessing && (
            <div className="absolute inset-0 flex flex-col items-center justify-center p-6 text-center bg-slate-950/80 pointer-events-none z-10">
              <Smartphone className="w-10 h-10 text-slate-600 mb-2 stroke-1" />
              <span className="text-xs font-semibold text-slate-300">Nenhum vídeo selecionado</span>
              <span className="text-[10px] text-slate-500 mt-1 max-w-[210px] leading-relaxed">
                Adicione vídeos na aba "Fila em Lote" ou teste um exemplo para ver a prévia ao vivo.
              </span>
            </div>
          )}
        </div>
      </div>

      {/* Video Transport Controls */}
      {activeItem && (
        <div className="w-full mt-4 flex items-center justify-between gap-3 text-xs bg-slate-950/60 p-2.5 rounded-xl border border-slate-800">

              className="px-3 py-1.5 rounded-lg bg-emerald-600 hover:bg-emerald-500 text-white font-bold flex items-center gap-1.5 transition shadow-sm"

              <Download className="w-3.5 h-3.5" />
              {activeItem.outputBlob?.type.includes('mp4') ? 'Baixar MP4' : 'Baixar Vídeo HD'}

};

/**
 * Synthesizes an AudioBuffer of specified duration for a given preset track
 */
export async function generateSyntheticTrackBuffer(
  audioCtx: AudioContext | OfflineAudioContext,
  trackId: string,
  targetDuration: number
): Promise<AudioBuffer | null> {
  if (trackId === 'none') return null;

  const sampleRate = audioCtx.sampleRate || 44100;
  const totalSamples = Math.ceil(targetDuration * sampleRate);
  const buffer = audioCtx.createBuffer(2, totalSamples, sampleRate);
  const left = buffer.getChannelData(0);
  const right = buffer.getChannelData(1);

  const selectedTrack = PRESET_TRACKS.find(t => t.id === trackId) || PRESET_TRACKS[1];
  const bpm = selectedTrack.bpm || 80;
  const beatSec = 60 / bpm;

  for (let i = 0; i < totalSamples; i++) {
    const t = i / sampleRate;
    let sampleL = 0;
    let sampleR = 0;

    if (trackId === 'lofi') {
      // Warm chords + soft vinyl crackle
      const chordTime = t % (beatSec * 4);
      const chordIndex = Math.floor(t / (beatSec * 4)) % 4;
      const chords = [
        [220, 261.6, 329.6, 392.0], // Am7
        [174.6, 220, 261.6, 329.6], // Fmaj7
        [261.6, 329.6, 392.0, 523.2], // Cmaj7
        [196.0, 246.9, 293.6, 392.0], // G7
      ];
      const activeChord = chords[chordIndex];
      const chordEnv = Math.exp(-chordTime * 0.8) * 0.12;

      for (let c = 0; c < activeChord.length; c++) {
        const freq = activeChord[c];
        const osc = Math.sin(2 * Math.PI * freq * t) + 0.3 * Math.sin(2 * Math.PI * freq * 2 * t);
        sampleL += osc * chordEnv;
        sampleR += (osc * 0.9) * chordEnv;
      }

      // Soft bass
      const bassFreq = activeChord[0] / 2;
      const bass = Math.sin(2 * Math.PI * bassFreq * t) * 0.15;
      sampleL += bass;
      sampleR += bass;

      // Vinyl noise
      const crackle = (Math.random() - 0.5) * 0.008;
      sampleL += crackle;
      sampleR += crackle;
    } else if (trackId === 'phonk') {
      // Punchy 808 kick + cowbell synth
      const beatProgress = t % beatSec;
      const kickEnv = Math.exp(-beatProgress * 9) * 0.25;
      const kickFreq = 55 + Math.exp(-beatProgress * 25) * 110;
      const kick = Math.sin(2 * Math.PI * kickFreq * t) * kickEnv;

      // Cowbell melody on 16th notes
      const sixteenth = t % (beatSec / 2);
      const bellStep = Math.floor(t / (beatSec / 2)) % 8;
      const bellFreqs = [587.3, 659.2, 587.3, 783.9, 880.0, 783.9, 659.2, 523.2];
      const bellEnv = Math.exp(-sixteenth * 12) * 0.1;
      const bell = (Math.sin(2 * Math.PI * bellFreqs[bellStep] * t) + 0.5 * Math.sin(2 * Math.PI * bellFreqs[bellStep] * 2.5 * t)) * bellEnv;

      sampleL += kick + bell * 0.8;
      sampleR += kick + bell * 0.9;
    } else if (trackId === 'suspense') {
      // Dramatic sub pulse + high tension violin drone
      const pulseTime = t % (beatSec * 2);
      const pulseEnv = Math.exp(-pulseTime * 3) * 0.22;
      const sub = Math.sin(2 * Math.PI * 48 * t) * pulseEnv;
      const tension = (Math.sin(2 * Math.PI * 440 * t) + Math.sin(2 * Math.PI * 445 * t)) * 0.04;
      const riser = Math.sin(2 * Math.PI * (200 + (t % 8) * 40) * t) * 0.03;

      sampleL += sub + tension + riser;
      sampleR += sub + tension - riser;
    } else if (trackId === 'trap') {
      // 808 Trap bounce & fast hi-hats
      const beatProgress = t % beatSec;
      const sub808 = Math.sin(2 * Math.PI * 42 * t) * Math.exp(-beatProgress * 2.5) * 0.25;
      const hatProgress = t % (beatSec / 4);
      const hat = (Math.random() - 0.5) * Math.exp(-hatProgress * 30) * 0.08;
      const snare = (Math.floor(t / beatSec) % 2 === 1) ? (Math.random() - 0.5) * Math.exp(-beatProgress * 10) * 0.15 : 0;

      sampleL += sub808 + hat + snare;
      sampleR += sub808 + hat * 0.9 + snare;
    } else {
      // Chill podcast atmosphere
      const padTime = t % 12;
      const pad = (Math.sin(2 * Math.PI * 130.81 * t) + Math.sin(2 * Math.PI * 196 * t) + Math.sin(2 * Math.PI * 246.94 * t)) * 0.06;
      const chime = Math.sin(2 * Math.PI * 523.25 * t) * Math.exp(-(t % 4) * 2) * 0.04;

      sampleL += pad + chime;
      sampleR += pad + chime * 0.8;
    }

    // Gentle master clamp
    left[i] = Math.max(-1, Math.min(1, sampleL));
    right[i] = Math.max(-1, Math.min(1, sampleR));
  }

  return buffer;
}

import { VideoQueueItem } from '../types/video';

/**
 * Creates a synthetic demo video using HTML Canvas and Web Audio directly in the browser
 */
export async function createSyntheticDemoVideo(): Promise<VideoQueueItem> {
  const canvas = document.createElement('canvas');
  canvas.width = 720;
  canvas.height = 1280;
  canvas.style.position = 'fixed';
  canvas.style.top = '-9999px';
  canvas.style.left = '-9999px';
  canvas.style.opacity = '0';
  canvas.style.pointerEvents = 'none';
  document.body.appendChild(canvas);

  const ctx = canvas.getContext('2d')!;

  // Pre-render frame 0
  ctx.fillStyle = '#0f172a';
  ctx.fillRect(0, 0, 720, 1280);

  const fps = 15;
  const durationSec = 1.5;
  const totalFrames = Math.round(fps * durationSec);

  const stream = canvas.captureStream(fps);

  // Add synthesized audio beep track safely only if AudioContext is active
  let audioCtx: AudioContext | null = null;
  let osc: OscillatorNode | null = null;
  try {
    const AudioCtxClass = window.AudioContext || (window as any).webkitAudioContext;
    if (AudioCtxClass) {
      audioCtx = new AudioCtxClass();
      if (audioCtx.state === 'suspended') {
        audioCtx.resume().catch(() => {});
      }
      if (audioCtx.state === 'running') {
        const dest = audioCtx.createMediaStreamDestination();
        osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(440, 0);
        gain.gain.setValueAtTime(0.02, 0);
        osc.connect(gain);
        gain.connect(dest);
        osc.start();
        stream.addTrack(dest.stream.getAudioTracks()[0]);
      }
    }
  } catch (_) {}

  const chunks: Blob[] = [];
  const selectedMime = MediaRecorder.isTypeSupported('video/webm;codecs=vp8,opus')
    ? 'video/webm;codecs=vp8,opus'
    : MediaRecorder.isTypeSupported('video/webm;codecs=vp8')
    ? 'video/webm;codecs=vp8'
    : MediaRecorder.isTypeSupported('video/webm')
    ? 'video/webm'
    : 'video/mp4';

  const recorder = new MediaRecorder(stream, {
    mimeType: selectedMime,
  });

  recorder.ondataavailable = (e) => {
    if (e.data && e.data.size > 0) chunks.push(e.data);
  };

  const recordingPromise = new Promise<Blob>((resolve) => {
    recorder.onstop = () => {
      try {
        if (osc) osc.stop();
        if (audioCtx && audioCtx.state !== 'closed') audioCtx.close();
      } catch (_) {}
      resolve(new Blob(chunks, { type: selectedMime }));
    };
  });

  recorder.start();

  let frame = 0;
  const renderFrame = () => {
    if (frame >= totalFrames) {
      recorder.stop();
      return;
    }

    const t = frame / fps;

    // Draw dynamic background
    const grad = ctx.createLinearGradient(0, 0, 720, 1280);
    grad.addColorStop(0, '#0f172a');
    grad.addColorStop(0.5, '#1e1b4b');
    grad.addColorStop(1, '#0284c7');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, 720, 1280);

    // Draw moving particles & shapes
    for (let i = 0; i < 20; i++) {
      const x = (Math.sin(t * 2 + i) * 0.4 + 0.5) * 720;
      const y = (Math.cos(t * 1.5 + i * 2) * 0.4 + 0.5) * 1280;
      const r = 20 + Math.sin(t * 4 + i) * 15;
      ctx.fillStyle = `hsla(${(i * 25 + t * 40) % 360}, 85%, 65%, 0.6)`;
      ctx.beginPath();
      ctx.arc(x, y, Math.max(5, r), 0, Math.PI * 2);
      ctx.fill();
    }

    // Top banner demo label
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 38px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('VÍDEO DE DEMONSTRAÇÃO', 360, 480);

    ctx.font = '24px sans-serif';
    ctx.fillStyle = '#38bdf8';
    ctx.fillText('WiFive Pro Studio · Teste em Tempo Real', 360, 530);

    // Countdown / Timer circle
    ctx.beginPath();
    ctx.arc(360, 680, 100, 0, Math.PI * 2);
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
    ctx.lineWidth = 12;
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(360, 680, 100, -Math.PI / 2, -Math.PI / 2 + (t / durationSec) * Math.PI * 2);
    ctx.strokeStyle = '#38bdf8';
    ctx.lineWidth = 12;
    ctx.stroke();

    ctx.font = 'bold 48px monospace';
    ctx.fillStyle = '#ffffff';
    ctx.fillText(`${(durationSec - t).toFixed(1)}s`, 360, 695);

    // Watermark to test edge cropping
    ctx.fillStyle = '#ef4444';
    ctx.fillRect(20, 20, 220, 40);
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 16px sans-serif';
    ctx.fillText('MARCA D’ÁGUA TOPO', 130, 46);

    ctx.fillStyle = '#eab308';
    ctx.fillRect(20, 1220, 260, 40);
    ctx.fillStyle = '#000000';
    ctx.fillText('LEGENDA ANTIGA BASE', 150, 1246);

    frame++;
    requestAnimationFrame(renderFrame);
  };

  renderFrame();

  const rawBlob = await recordingPromise;

  // Generate thumbnail before canvas is removed
  const thumbCanvas = document.createElement('canvas');
  thumbCanvas.width = 180;
  thumbCanvas.height = 320;
  const tCtx = thumbCanvas.getContext('2d')!;
  tCtx.drawImage(canvas, 0, 0, 180, 320);
  const thumbnailUrl = thumbCanvas.toDataURL('image/jpeg');

  if (canvas.parentNode) {
    canvas.parentNode.removeChild(canvas);
  }

  // Inject duration if WebM to prevent player freezing
  let finalBlob = rawBlob;
  if (rawBlob.type.includes('webm')) {
    finalBlob = await fixWebmDuration(rawBlob, durationSec);
  }

  const url = URL.createObjectURL(finalBlob);
  const ext = finalBlob.type.includes('mp4') ? 'mp4' : 'webm';
  const file = new File([finalBlob], `demo_corte_viral.${ext}`, { type: finalBlob.type });

  return {
    id: `demo_${Date.now()}`,
    file,
    url,
    name: `demo_corte_viral.${ext}`,
    duration: durationSec,
    width: 720,
    height: 1280,
    thumbnailUrl,
    trim: {
      startTime: 0,
      endTime: durationSec,
    },
    status: 'idle',
    progress: 0,
    outputBlob: null,
    outputUrl: null,
    errorMessage: null,
  };
}

import JSZip from 'jszip';
import { VideoQueueItem } from '../types/video';

export async function downloadAllVideosAsZip(
  items: VideoQueueItem[],
  onProgress?: (percent: number) => void
): Promise<void> {
  const completed = items.filter((item) => item.status === 'done' && item.outputBlob);
  if (completed.length === 0) return;

  const zip = new JSZip();
  const folder = zip.folder('cortes_wifive_pro_hd');

  completed.forEach((item, index) => {
    const cleanName = item.name.replace(/\.[^/.]+$/, '').replace(/[^a-zA-Z0-9_-]/g, '_');
    const ext = item.outputBlob?.type.includes('mp4') ? 'mp4' : 'webm';
    const fileName = `${String(index + 1).padStart(2, '0')}_${cleanName}_wifive_pro.${ext}`;
    if (folder && item.outputBlob) {
      folder.file(fileName, item.outputBlob);
    }
  });

  const zipBlob = await zip.generateAsync({ type: 'blob' }, (metadata) => {
    if (onProgress) {
      onProgress(Math.round(metadata.percent));
    }
  });

  const url = URL.createObjectURL(zipBlob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `cortes_wifive_pro_hd_${new Date().toISOString().slice(0, 10)}.zip`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(url), 10000);
}

export async function downloadLinuxPackage(): Promise<void> {
  const fileName = 'wifive-pro-studio-linux.zip';
  try {
    const res = await fetch(`/${fileName}`);
    const ct = res.headers.get('content-type') || '';
    if (res.ok && !ct.includes('text/html')) {
      const blob = await res.blob();
      if (blob.size > 5000) {
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = fileName;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        setTimeout(() => URL.revokeObjectURL(url), 10000);
        return;
      }
    }
  } catch (err) {
    console.warn('Direct fetch failed, falling back to embedded package:', err);
  }

  // Fallback to embedded Base64 bundle (guarantees download even inside strict sandboxed iframes)
  const { LINUX_PACKAGE_BASE64 } = await import('./linuxPackageBase64');
  triggerBase64Download(LINUX_PACKAGE_BASE64, fileName, 'application/zip');
}

function triggerBase64Download(base64: string, fileName: string, mimeType: string): void {
  const byteCharacters = atob(base64);
  const byteArray = new Uint8Array(byteCharacters.length);
  for (let i = 0; i < byteCharacters.length; i++) {
    byteArray[i] = byteCharacters.charCodeAt(i);
  }
  const blob = new Blob([byteArray], { type: mimeType });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = fileName;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  setTimeout(() => URL.revokeObjectURL(url), 10000);
}

import { FrameStyle, ProfileBrand, VisualEffects } from '../types/video';

export interface FrameLayoutSlot {
  top: number; // percentage of canvas height
  bottom: number; // percentage of canvas height
  left: number; // percentage of canvas width
  right: number; // percentage of canvas width
}

/**
 * Returns the inner viewport area reserved for the video according to frame style
 */
export function getFrameLayoutSlot(style: FrameStyle): FrameLayoutSlot {
  switch (style) {
    case 'post':
      return { top: 0.17, bottom: 0.08, left: 0.03, right: 0.03 };
    case 'noticia':
      return { top: 0.14, bottom: 0.14, left: 0.02, right: 0.02 };
    case 'meme':
      return { top: 0.18, bottom: 0.06, left: 0.02, right: 0.02 };
    case 'podcast':
      return { top: 0.14, bottom: 0.12, left: 0.03, right: 0.03 };
    case 'neon':
      return { top: 0.06, bottom: 0.06, left: 0.04, right: 0.04 };
    case 'gradiente':
      return { top: 0.09, bottom: 0.09, left: 0.03, right: 0.03 };
    case 'minimal':
      return { top: 0.05, bottom: 0.05, left: 0.03, right: 0.03 };
    case 'custom':
    case 'none':
    default:
      return { top: 0, bottom: 0, left: 0, right: 0 };
  }
}

/**
 * Renders the background (e.g. blurred video or solid dark color)
 */
export function renderBackground(
  ctx: CanvasRenderingContext2D,
  width: number,
  height: number,
  videoEl: HTMLVideoElement | null,
  effects: VisualEffects,
  fitMode: 'contain-blur' | 'contain-solid' | 'cover',
  backdropColor: string
) {
  ctx.save();

  if (fitMode === 'contain-blur' && videoEl && videoEl.readyState >= 2) {
    // Fill with blurred video using hardware-efficient blur radius
    const blurRadius = Math.min(10, Math.max(3, Math.round((effects.blurBackground || 25) * 0.4)));
    ctx.filter = `blur(${blurRadius}px) brightness(0.65) saturate(1.2)`;
    // Draw enlarged to cover canvas
    const scale = Math.max(width / videoEl.videoWidth, height / videoEl.videoHeight) * 1.25;
    const bgW = videoEl.videoWidth * scale;
    const bgH = videoEl.videoHeight * scale;
    ctx.drawImage(videoEl, (width - bgW) / 2, (height - bgH) / 2, bgW, bgH);
    ctx.filter = 'none';

    // Dark gradient overlay for readability
    const bgGrad = ctx.createLinearGradient(0, 0, 0, height);
    bgGrad.addColorStop(0, 'rgba(5, 10, 24, 0.55)');
    bgGrad.addColorStop(0.5, 'rgba(5, 10, 24, 0.35)');
    bgGrad.addColorStop(1, 'rgba(5, 10, 24, 0.65)');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, width, height);
  } else {
    // Solid backdrop
    ctx.fillStyle = backdropColor || '#0a0d18';
    ctx.fillRect(0, 0, width, height);
  }

  ctx.restore();
}

/**
 * Draws avatar in circle, falling back to clean initials
 */
function drawAvatar(
  ctx: CanvasRenderingContext2D,
  img: HTMLImageElement | null,
  cx: number,
  cy: number,
  r: number,
  name: string
) {
  ctx.save();
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.closePath();
  ctx.clip();

  if (img && img.complete && img.naturalWidth > 0) {
    ctx.drawImage(img, cx - r, cy - r, r * 2, r * 2);
  } else {
    // Initials fallback
    ctx.fillStyle = '#2563eb';
    ctx.fillRect(cx - r, cy - r, r * 2, r * 2);
    ctx.fillStyle = '#ffffff';
    ctx.font = `bold ${Math.round(r * 0.9)}px sans-serif`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    const initial = (name ? name.trim()[0] : 'W').toUpperCase();
    ctx.fillText(initial, cx, cy);
  }

  ctx.restore();

  // Avatar border
  ctx.save();
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.strokeStyle = '#ffffff';
  ctx.lineWidth = 2.5;
  ctx.stroke();
  ctx.restore();
}

/**
 * Verified check badge
 */
function drawBadge(
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  size: number,
  type: 'blue' | 'gold' | 'none'
) {
  if (type === 'none') return;
  ctx.save();
  ctx.beginPath();
  const r = size / 2;
  ctx.arc(x + r, y, r, 0, Math.PI * 2);
  ctx.fillStyle = type === 'gold' ? '#eab308' : '#1d9bf0';
  ctx.fill();

  // White checkmark
  ctx.strokeStyle = '#ffffff';
  ctx.lineWidth = 2.5;
  ctx.beginPath();
  ctx.moveTo(x + r * 0.55, y);
  ctx.lineTo(x + r * 0.9, y + r * 0.35);
  ctx.lineTo(x + r * 1.5, y - r * 0.35);
  ctx.stroke();
  ctx.restore();
}

/**
 * Canvas rounded rectangle helper
 */
function roundRect(
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  w: number,
  h: number,
  radius: number
) {
  ctx.beginPath();
  ctx.moveTo(x + radius, y);
  ctx.arcTo(x + w, y, x + w, y + h, radius);
  ctx.arcTo(x + w, y + h, x, y + h, radius);
  ctx.arcTo(x, y + h, x, y, radius);
  ctx.arcTo(x, y, x + w, y, radius);
  ctx.closePath();
}

/**
 * Text wrapper for meme and title boxes
 */
function wrapText(
  ctx: CanvasRenderingContext2D,
  text: string,
  x: number,
  y: number,
  maxWidth: number,
  lineHeight: number
) {
  const words = text.split(' ');
  let line = '';
  const lines: string[] = [];

  for (let n = 0; n < words.length; n++) {
    const testLine = line + words[n] + ' ';
    const metrics = ctx.measureText(testLine);
    if (metrics.width > maxWidth && n > 0) {
      lines.push(line.trim());
      line = words[n] + ' ';
    } else {
      line = testLine;
    }
  }
  lines.push(line.trim());

  const totalH = lines.length * lineHeight;
  const startY = y - totalH / 2 + lineHeight / 2;

  for (let k = 0; k < lines.length; k++) {
    ctx.fillText(lines[k], x, startY + k * lineHeight);
  }
}

export interface CropSettings {
  cropTop: number; // percentage 0-50
  cropBottom: number; // percentage 0-50
  cropLeft: number; // percentage 0-50
  cropRight: number; // percentage 0-50
  scale: number; // 0.5 to 2.0
  positionY: number; // percentage 0 to 100 (50 is center)
  positionX: number; // percentage 0 to 100 (50 is center)
}

  endTime: number; // seconds

/**
 * Calculates the source rectangle on the original video after crop percentages are applied.
 */
export function calculateSourceCropRect(
  videoWidth: number,
  videoHeight: number,
  crop: CropSettings
): { sx: number; sy: number; sw: number; sh: number } {
  const leftPx = (crop.cropLeft / 100) * videoWidth;
  const rightPx = (crop.cropRight / 100) * videoWidth;
  const topPx = (crop.cropTop / 100) * videoHeight;
  const bottomPx = (crop.cropBottom / 100) * videoHeight;

  const sx = Math.max(0, leftPx);
  const sy = Math.max(0, topPx);
  const sw = Math.max(1, videoWidth - leftPx - rightPx);
  const sh = Math.max(1, videoHeight - topPx - bottomPx);

  return { sx, sy, sw, sh };
}

/**
 * Calculates the destination placement for the video inside target canvas.
 */
export function calculateDestPlacement(
  canvasWidth: number,
  canvasHeight: number,
  croppedAspect: number,
  frameVideoSlot: { top: number; bottom: number; left: number; right: number },
  scale: number,
  positionX: number,
  positionY: number
): { dx: number; dy: number; dw: number; dh: number } {
  // Available slot within canvas
  const slotWidth = canvasWidth * (1 - frameVideoSlot.left - frameVideoSlot.right);
  const slotHeight = canvasHeight * (1 - frameVideoSlot.top - frameVideoSlot.bottom);
  const slotCenterX = canvasWidth * frameVideoSlot.left + slotWidth / 2;
  const slotCenterY = canvasHeight * frameVideoSlot.top + slotHeight / 2;

  let baseWidth = slotWidth;
  let baseHeight = slotWidth / croppedAspect;

  if (baseHeight > slotHeight) {
    baseHeight = slotHeight;
    baseWidth = slotHeight * croppedAspect;
  }

  const dw = baseWidth * scale;
  const dh = baseHeight * scale;

  // Offset based on positionX and positionY (50 is center)
  const offsetX = ((positionX - 50) / 100) * slotWidth;
  const offsetY = ((positionY - 50) / 100) * slotHeight;

  const dx = slotCenterX - dw / 2 + offsetX;
  const dy = slotCenterY - dh / 2 + offsetY;

  return { dx, dy, dw, dh };
}

/**
 * Sanitizes trim values to ensure valid start and end bounds
 */
export function validateTrimBounds(
  startTime: number,
  endTime: number,
  totalDuration: number
): { startTime: number; endTime: number; duration: number } {
  const safeStart = Math.max(0, Math.min(startTime, totalDuration - 0.1));
  const safeEnd = Math.max(safeStart + 0.1, Math.min(endTime, totalDuration));
  return {
    startTime: safeStart,
    endTime: safeEnd,
    duration: Math.max(0.1, safeEnd - safeStart),
  };
}

/**
 * Formats time in seconds to mm:ss.ms
 */
export function formatTime(seconds: number): string {
  if (isNaN(seconds) || seconds < 0) return '00:00';
  const mins = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);
  const ms = Math.floor((seconds % 1) * 10);
  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}.${ms}`;
}

/**
 * Returns supported MIME type for MediaRecorder, preferring standard formats
 */
export function getSupportedMimeType(): { mimeType: string; extension: string } {
  const candidateMimes = [
    { mime: 'video/webm;codecs=vp8,opus', ext: 'webm' },
    { mime: 'video/webm;codecs=vp8', ext: 'webm' },
    { mime: 'video/webm;codecs=h264,opus', ext: 'webm' },
    { mime: 'video/webm', ext: 'webm' },
    { mime: 'video/mp4;codecs=avc1.42E01E,mp4a.40.2', ext: 'mp4' },
    { mime: 'video/mp4', ext: 'mp4' },
  ];

  if (typeof MediaRecorder !== 'undefined') {
    for (const item of candidateMimes) {
      if (MediaRecorder.isTypeSupported(item.mime)) {
        return { mimeType: item.mime, extension: item.ext };
      }
    }
  }

  return { mimeType: 'video/webm', extension: 'webm' };
}

/**
 * Injects or updates exact duration in WebM EBML header to prevent player freezing/hanging
 */
export async function fixWebmDuration(blob: Blob, durationSec: number): Promise<Blob> {
  return new Promise((resolve) => {
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const buffer = reader.result as ArrayBuffer;
        const u8 = new Uint8Array(buffer);

        // Find Segment Info element: 0x15, 0x49, 0xA9, 0x66
        let infoPos = -1;
        for (let i = 0; i < Math.min(u8.length - 4, 4096); i++) {
          if (u8[i] === 0x15 && u8[i + 1] === 0x49 && u8[i + 2] === 0xa9 && u8[i + 3] === 0x66) {
            infoPos = i;
            break;
          }
        }

        if (infoPos === -1) {
          return resolve(blob);
        }

        const durationMs = durationSec * 1000;

        // Check if Duration element (0x44, 0x89) already exists inside Info
        let durPos = -1;
        for (let i = infoPos + 4; i < Math.min(u8.length - 2, infoPos + 300); i++) {
          if (u8[i] === 0x44 && u8[i + 1] === 0x89) {
            durPos = i;
            break;
          }
        }

        if (durPos !== -1) {
          const lenByte = u8[durPos + 2];
          const durLen = lenByte & 0x7f;
          if (durLen === 4 || durLen === 8) {
            const dataView = new DataView(buffer, durPos + 3, durLen);
            if (durLen === 4) {
              dataView.setFloat32(0, durationMs);
            } else {
              dataView.setFloat64(0, durationMs);
            }
            return resolve(new Blob([buffer], { type: blob.type }));
          }
        }

        // Duration element: ID [0x44, 0x89], length 8 [0x88], float64 value
        const durBytes = new Uint8Array(11);
        durBytes[0] = 0x44;
        durBytes[1] = 0x89;
        durBytes[2] = 0x88;
        const dv = new DataView(durBytes.buffer, 3, 8);
        dv.setFloat64(0, durationMs);

        // Determine Info length header size
        const lenByte = u8[infoPos + 4];
        let headerLen = 5;
        let oldInfoDataLen = lenByte & 0x7f;
        if ((lenByte & 0x80) === 0 && (lenByte & 0x40) !== 0) {
          headerLen = 6;
          oldInfoDataLen = ((lenByte & 0x3f) << 8) | u8[infoPos + 5];
        }

        const newInfoDataLen = oldInfoDataLen + 11;
        const newBuffer = new Uint8Array(u8.length + 11);

        const insertPos = infoPos + headerLen;
        newBuffer.set(u8.subarray(0, insertPos), 0);

        if (headerLen === 5) {
          newBuffer[infoPos + 4] = 0x80 | newInfoDataLen;
        } else {
          newBuffer[infoPos + 4] = 0x40 | (newInfoDataLen >> 8);
          newBuffer[infoPos + 5] = newInfoDataLen & 0xff;
        }

        newBuffer.set(durBytes, insertPos);
        newBuffer.set(u8.subarray(insertPos), insertPos + 11);

        resolve(new Blob([newBuffer.buffer], { type: blob.type }));
      } catch (_) {
        resolve(blob);
      }
    };
    reader.onerror = () => resolve(blob);
    reader.readAsArrayBuffer(blob);
  });
}

/**
 * Calculates canvas dimensions based on resolution and aspect ratio
 */
export function getExportResolution(
  resolution: '720p' | '1080p' | '1440p',
  aspectRatio: '9:16' | '1:1' | '16:9'
): { width: number; height: number } {
  if (aspectRatio === '9:16') {
    switch (resolution) {
      case '1440p':
        return { width: 1440, height: 2560 };
      case '720p':
        return { width: 720, height: 1280 };
      case '1080p':
      default:
        return { width: 1080, height: 1920 };
    }
  } else if (aspectRatio === '1:1') {
    switch (resolution) {
      case '1440p':
        return { width: 1440, height: 1440 };
      case '720p':
        return { width: 720, height: 720 };
      case '1080p':
      default:
        return { width: 1080, height: 1080 };
    }
  } else {
    // 16:9 landscape
    switch (resolution) {
      case '1440p':
        return { width: 2560, height: 1440 };
      case '720p':
        return { width: 1280, height: 720 };
      case '1080p':
      default:
        return { width: 1920, height: 1080 };
    }
  }
}

/**
 * Processes a single video with all trimming, cropping, frame overlay, effects, speed & audio mixing
 */
export async function processVideoItem(
  item: VideoQueueItem,
  frameStyle: FrameStyle,
  brand: ProfileBrand,
  effects: VisualEffects,
  audioSettings: AudioSettings,
  cropSettings: VideoCropSettings,
  exportSettings: ExportSettings,
  onProgress: ProcessorProgressCallback,
  signal?: AbortSignal
): Promise<{ outputBlob: Blob; outputUrl: string }> {
  return new Promise(async (resolve, reject) => {
    let cleanupRun = false;
    let animFrameId: number | null = null;
    let backgroundIntervalId: any = null;
    let audioCtx: AudioContext | null = null;
    let mediaRecorder: MediaRecorder | null = null;
    let canvas: HTMLCanvasElement | null = null;

    const cleanup = () => {
      if (cleanupRun) return;
      cleanupRun = true;
      if (animFrameId) cancelAnimationFrame(animFrameId);
      if (backgroundIntervalId) clearInterval(backgroundIntervalId);
      if (audioCtx && audioCtx.state !== 'closed') {
        try {
          audioCtx.close();
        } catch (_) {}
      }
      if (videoEl) {
        videoEl.pause();
        videoEl.removeAttribute('src');
        videoEl.load();
      }
      if (canvas && canvas.parentNode) {
        try {
          canvas.parentNode.removeChild(canvas);
        } catch (_) {}
      }
    };

    if (signal?.aborted) {
      cleanup();
      return reject(new Error('Processamento cancelado pelo usuário'));
    }

    signal?.addEventListener('abort', () => {
      if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.stop();
      }
      cleanup();
      reject(new Error('Processamento cancelado pelo usuário'));
    });

    onProgress(5, 'Carregando vídeo e preparando decodificador...');

    // 1. Prepare video element
    const videoEl = document.createElement('video');
    videoEl.crossOrigin = 'anonymous';
    videoEl.playsInline = true;
    videoEl.muted = true; // Muted to avoid browser autoplay restrictions
    videoEl.preload = 'auto';
    videoEl.src = item.url;

    try {
      await new Promise<void>((res, rej) => {
        videoEl.onloadedmetadata = () => res();
        videoEl.onerror = () => rej(new Error('Erro ao carregar o arquivo de vídeo. Formato incompatível.'));
      });
    } catch (err: any) {
      cleanup();
      return reject(err);
    }

    const videoDur = videoEl.duration || 10;
    const startTime = Math.max(0, Math.min(item.trim.startTime, videoDur - 0.2));
    const endTime = Math.min(item.trim.endTime > startTime ? item.trim.endTime : videoDur, videoDur);
    const clipDuration = Math.max(0.5, endTime - startTime);

    onProgress(12, 'Configurando resolução e efeitos...');

    // 2. Prepare Canvas - Must be attached to DOM for active Chromium compositor capture
    const { width: targetW, height: targetH } = getExportResolution(
      exportSettings.resolution,
      exportSettings.aspectRatio
    );

    canvas = document.createElement('canvas');
    canvas.width = targetW;
    canvas.height = targetH;
    canvas.style.position = 'fixed';
    canvas.style.top = '-9999px';
    canvas.style.left = '-9999px';
    canvas.style.opacity = '0';
    canvas.style.pointerEvents = 'none';
    document.body.appendChild(canvas);

    const ctx = canvas.getContext('2d', { alpha: false, desynchronized: true });
    if (!ctx) {
      cleanup();
      return reject(new Error('Falha ao inicializar o contexto Canvas 2D'));
    }

    // Pre-draw initial frame to ensure captureStream initializes with valid dimensions
    ctx.fillStyle = cropSettings.backdropColor || '#0a0d18';
    ctx.fillRect(0, 0, targetW, targetH);

    // Pre-load images if any
    let avatarImg: HTMLImageElement | null = null;
    if (brand.avatarUrl) {
      avatarImg = new Image();
      avatarImg.crossOrigin = 'anonymous';
      avatarImg.src = brand.avatarUrl;
      await new Promise((r) => {
        avatarImg!.onload = r;
        avatarImg!.onerror = r;
      });
    }

    let customFrameImg: HTMLImageElement | null = null;
    if (frameStyle === 'custom' && brand.customFrameUrl) {
      customFrameImg = new Image();
      customFrameImg.crossOrigin = 'anonymous';
      customFrameImg.src = brand.customFrameUrl;
      await new Promise((r) => {
        customFrameImg!.onload = r;
        customFrameImg!.onerror = r;
      });
    }

    onProgress(20, 'Sintetizando e mixando áudio...');

    // 3. Audio Setup via Web Audio with zero-memory direct streaming
    const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
    audioCtx = new AudioContextClass();
    if (audioCtx.state === 'suspended') {
      try {
        await audioCtx.resume();
      } catch (_) {}
    }
    const dest = audioCtx.createMediaStreamDestination();

    let hasAudioTrack = false;
    let videoAudioSource: MediaElementAudioSourceNode | null = null;
    try {
      videoAudioSource = audioCtx.createMediaElementSource(videoEl);
      const originalGain = audioCtx.createGain();
      originalGain.gain.value = audioSettings.originalVolume;
      videoAudioSource.connect(originalGain);
      originalGain.connect(dest);
      hasAudioTrack = true;
    } catch (e) {
      console.warn('MediaElementAudioSource could not be attached directly:', e);
    }

    // Background Music Track
    let bgmBuffer: AudioBuffer | null = null;
    if (audioSettings.customAudioUrl) {
      try {
        const audioFetch = await fetch(audioSettings.customAudioUrl);
        const arrayBuf = await audioFetch.arrayBuffer();
        bgmBuffer = await audioCtx.decodeAudioData(arrayBuf);
      } catch (err) {
        console.warn('Failed to decode custom audio file, falling back', err);
      }
    } else if (audioSettings.bgmTrackId !== 'none') {
      try {
        bgmBuffer = await generateSyntheticTrackBuffer(audioCtx, audioSettings.bgmTrackId, clipDuration + 2);
      } catch (err) {
        console.warn('Failed to generate synthetic background audio', err);
      }
    }

    let bgmSource: AudioBufferSourceNode | null = null;
    if (bgmBuffer) {
      bgmSource = audioCtx.createBufferSource();
      bgmSource.buffer = bgmBuffer;
      bgmSource.loop = true;
      const bgmGain = audioCtx.createGain();
      bgmGain.gain.value = audioSettings.bgmVolume;
      bgmSource.connect(bgmGain);
      bgmGain.connect(dest);
      hasAudioTrack = true;
    }

    // 4. Stream and MediaRecorder
    onProgress(28, 'Iniciando captura em alta definição...');
    const canvasStream = canvas.captureStream(exportSettings.fps || 30);

    // Merge audio tracks into stream only if audio is available
    const audioTracks = dest.stream.getAudioTracks();
    if (audioTracks.length > 0 && hasAudioTrack) {
      canvasStream.addTrack(audioTracks[0]);
    }

    const { mimeType } = getSupportedMimeType();
    const bps = (exportSettings.bitrateMbps || 8) * 1000 * 1000;

    const chunks: Blob[] = [];
    try {
      mediaRecorder = new MediaRecorder(canvasStream, {
        mimeType: MediaRecorder.isTypeSupported(mimeType) ? mimeType : undefined,
        videoBitsPerSecond: bps,
      });
    } catch (e) {
      mediaRecorder = new MediaRecorder(canvasStream);
    }

    mediaRecorder.ondataavailable = (e) => {
      if (e.data && e.data.size > 0) {
        chunks.push(e.data);
      }
    };

    mediaRecorder.onerror = (e) => {
      cleanup();
      reject(new Error(`Erro no gravador de mídia: ${(e as any).error?.message || 'Falha de encoding'}`));
    };

    const completionPromise = new Promise<{ outputBlob: Blob; outputUrl: string }>((res) => {
      mediaRecorder!.onstop = async () => {
        onProgress(95, 'Limpando metadados e finalizando com taxa estável...');
        const rawBlob = new Blob(chunks, { type: mediaRecorder?.mimeType || mimeType });
        
        // Inject accurate duration if WebM to prevent player freezing/hanging on PC
        let finalBlob = rawBlob;
        if (rawBlob.type.includes('webm')) {
          finalBlob = await fixWebmDuration(rawBlob, clipDuration);
        }

        const finalUrl = URL.createObjectURL(finalBlob);
        cleanup();
        res({ outputBlob: finalBlob, outputUrl: finalUrl });
      };
    });

    // 5. Seek video to start with safe timeout (never hangs if already at startTime)
    if (Math.abs(videoEl.currentTime - startTime) > 0.05) {
      await new Promise<void>((r) => {
        let done = false;
        const onDone = () => {
          if (!done) {
            done = true;
            videoEl.onseeked = null;
            r();
          }
        };
        videoEl.onseeked = onDone;
        videoEl.currentTime = startTime;
        setTimeout(onDone, 600);
      });
    } else {
      videoEl.currentTime = startTime;
    }

    // Apply speed
    videoEl.playbackRate = exportSettings.speed || 1.0;
    if ('preservesPitch' in videoEl) {
      (videoEl as any).preservesPitch = true;
    }
    videoEl.loop = false;

    // Start recording with 250ms chunks for smooth memory flush
    mediaRecorder.start(250);

    if (bgmSource) {
      bgmSource.start(0);
    }

    try {
      videoEl.muted = false;
      await videoEl.play();
    } catch (_) {
      videoEl.muted = true;
      await videoEl.play().catch(() => {});
    }

    const slot = getFrameLayoutSlot(frameStyle);
    const activeCrop = {
      cropTop: cropSettings.cropTop,
      cropBottom: cropSettings.cropBottom,
      cropLeft: cropSettings.cropLeft,
      cropRight: cropSettings.cropRight,
      scale: cropSettings.scale,
      positionX: cropSettings.positionX,
      positionY: cropSettings.positionY,
    };

    // Check if visual filter string is non-default to bypass heavy Skia CPU filter overhead in 1080p
    const hasCustomFilters =
      effects.brightness !== 1 ||
      effects.contrast !== 1 ||
      effects.saturation !== 1 ||
      effects.sharpen;

    let filterStr = '';
    if (hasCustomFilters) {
      filterStr = `brightness(${effects.brightness}) contrast(${effects.contrast}) saturate(${effects.saturation})`;
      if (effects.sharpen) {
        filterStr += ' contrast(1.15)';
      }
    }

    // Progress throttling: update at most every 150ms or when progress changes significantly
    let lastProgressTime = 0;
    let lastProgressPct = -1;
    const sendThrottledProgress = (pct: number, statusText: string, force = false) => {
      const now = performance.now();
      if (force || pct >= 95 || pct !== lastProgressPct && (pct - lastProgressPct >= 1 || now - lastProgressTime >= 150)) {
        lastProgressTime = now;
        lastProgressPct = pct;
        onProgress(pct, statusText);
      }
    };

    // Hard ceiling watchdog so processing NEVER hangs
    const maxProcessingMs = Math.max(30000, ((clipDuration / (exportSettings.speed || 1)) * 3.5) * 1000);
    const hardTimeoutId = setTimeout(() => {
      console.warn('Watchdog: Finalizando stream com segurança.');
      if (mediaRecorder && mediaRecorder.state === 'recording') {
        try {
          mediaRecorder.requestData();
        } catch (_) {}
        mediaRecorder.stop();
      }
    }, maxProcessingMs);

    let isDrawing = false;
    const drawFrame = () => {
      if (cleanupRun || signal?.aborted || isDrawing) {
        return;
      }
      isDrawing = true;

      try {
        const curr = videoEl.currentTime;
        const elapsed = Math.max(0, curr - startTime);
        const pct = Math.min(94, 30 + Math.round((elapsed / clipDuration) * 64));
        sendThrottledProgress(pct, `Renderizando 1080p: ${elapsed.toFixed(1)}s / ${clipDuration.toFixed(1)}s (${pct}%)`);

        const isCompleted =
          curr >= endTime - 0.04 ||
          videoEl.ended ||
          elapsed >= clipDuration - 0.04;

        if (isCompleted) {
          clearTimeout(hardTimeoutId);
          if (mediaRecorder && mediaRecorder.state === 'recording') {
            try {
              mediaRecorder.requestData();
            } catch (_) {}
            mediaRecorder.stop();
          }
          return;
        }

        // 1. Draw dynamic background
        renderBackground(
          ctx,
          targetW,
          targetH,
          videoEl,
          effects,
          cropSettings.fitMode,
          cropSettings.backdropColor
        );

        // 2. Compute cropped video source and destination
        const srcRect = calculateSourceCropRect(videoEl.videoWidth || targetW, videoEl.videoHeight || targetH, activeCrop);
        const croppedAspect = srcRect.sw / srcRect.sh;
        const destRect = calculateDestPlacement(
          targetW,
          targetH,
          croppedAspect,
          slot,
          activeCrop.scale,
          activeCrop.positionX,
          activeCrop.positionY
        );

        // 3. Draw video with effects (bypass filter if standard for 10x faster 1080p performance)
        ctx.save();
        if (hasCustomFilters) {
          ctx.filter = filterStr;
        }

        ctx.drawImage(
          videoEl,
          srcRect.sx,
          srcRect.sy,
          srcRect.sw,
          srcRect.sh,
          destRect.dx,
          destRect.dy,
          destRect.dw,
          destRect.dh
        );
        ctx.restore();

        // Optional vignette
        if (effects.vignette) {
          ctx.save();
          const vigGrad = ctx.createRadialGradient(
            targetW / 2,
            targetH / 2,
            targetW * 0.35,
            targetW / 2,
            targetH / 2,
            targetW * 0.75
          );
          vigGrad.addColorStop(0, 'rgba(0,0,0,0)');
          vigGrad.addColorStop(1, 'rgba(0,0,0,0.55)');
          ctx.fillStyle = vigGrad;
          ctx.fillRect(0, 0, targetW, targetH);
          ctx.restore();
        }

        // 4. Draw branded frame overlay
        renderFrameOverlay(
          ctx,
          targetW,
          targetH,
          frameStyle,
          brand,
          avatarImg,
          customFrameImg,
          elapsed,
          item.customHeadline
        );
      } finally {
        isDrawing = false;
      }
    };

    // Unified non-competing rendering clock:
    // When tab is active, requestAnimationFrame paces at ~30 FPS;
    // When tab is in background, setInterval(33ms) takes over seamlessly.
    const targetIntervalMs = 1000 / (exportSettings.fps || 30);
    let lastRenderTimestamp = 0;

    const onAnimFrame = (timestamp: number) => {
      if (cleanupRun) return;
      if (timestamp - lastRenderTimestamp >= targetIntervalMs - 4) {
        lastRenderTimestamp = timestamp;
        drawFrame();
      }
      if (!cleanupRun) {
        animFrameId = requestAnimationFrame(onAnimFrame);
      }
    };
    animFrameId = requestAnimationFrame(onAnimFrame);

    // Fallback clock ONLY executes if requestAnimationFrame has been throttled (>45ms without a frame)
    backgroundIntervalId = setInterval(() => {
      if (cleanupRun) {
        clearInterval(backgroundIntervalId);
        return;
      }
      const now = performance.now();
      if (now - lastRenderTimestamp >= 45) {
        lastRenderTimestamp = now;
        drawFrame();
      }
    }, 33);

    try {
      const result = await completionPromise;
      resolve(result);
    } catch (err) {
      cleanup();
      reject(err);
    }
  });
}
